    <div class="task"><input type="checkbox" id="task1" /><label for="task1">Started 1 hard task?</label></div>

    <div class="task"><input type="checkbox" id="task2" /><label for="task2">Stayed focused for 25 mins?</label></div>

    <div class="task"><input type="checkbox" id="task3" /><label for="task3">Avoided distractions?</label></div>

    <div class="task"><input type="checkbox" id="task4" /><label for="task4">Finished at least one task?</label></div>

    <div class="task"><input type="checkbox" id="task5" /><label for="task5">Night reflection done?</label></div>


    <div class="streak">Streak: <span id="streakCount">0</span> days</div>

    <button class="reset-btn" onclick="resetTracker()">Reset</button>

  </div>


  <script>

    const checkboxes = document.querySelectorAll("input[type='checkbox']");

    const streakDisplay = document.getElementById("streakCount");

    let streak = parseInt(localStorage.getItem("disciplineStreak")) || 0;

    let completed = false;


    // Load previous checkboxes

    checkboxes.forEach((box, index) => {

      const saved = localStorage.getItem("task" + index);

      if (saved === "true") box.checked = true;

    });


    function saveProgress() {

      let allChecked = true;

      checkboxes.forEach((box, index) => {

        localStorage.setItem("task" + index, box.checked);

        if (!box.checked) allChecked = false;

      });

      if (allChecked && !completed) {

        streak++;

        localStorage.setItem("disciplineStreak", streak);

        streakDisplay.textContent = streak;

        completed = true;

      }

    }


    function resetTracker() {

      checkboxes.forEach((box, index) => {

        box.checked = false;

        localStorage.setItem("task" + index, false);

      });

      streak = 0;

      localStorage.setItem("disciplineStreak", 0);

      streakDisplay.textContent = "0";

      completed = false;

    }


    checkboxes.forEach(box => box.addEventListener("change", saveProgress));

    streakDisplay.textContent = streak;

  </script>
